Phase 2 · Design

Three tables, two foreign keys: one category has many destinations, one destination has many reviews.

TableStores
categoriesInterest types: Beach, Hill Station, Heritage, Adventure, Spiritual, Wildlife
destinationsPlace, state, category, budget level, best season, ideal days, description, image
reviewsTraveller name, rating 1 to 5, comment per destination

How recommendation works: the user picks interest, budget and season. The database filters matching destinations and ranks them by average rating, then by number of reviews. Draw the ER diagram in your record; you will add a column in Phase 12.

Phase 3 · Create tables (DDL)

Supabase → SQL Editor → New query, paste and Run.

-- database/01_tables.sql
create table categories (
  id   bigint generated always as identity primary key,
  name text not null unique
);
create table destinations (
  id          bigint generated always as identity primary key,
  name        text not null,
  state       text not null,
  category_id bigint references categories(id) on delete set null,
  budget      text not null check (budget in ('Low','Medium','High')),
  best_season text not null check (best_season in ('Winter','Summer','Monsoon')),
  ideal_days  int  check (ideal_days > 0),
  description text,
  image_url   text,
  created_at  timestamptz default now()
);
create table reviews (
  id             bigint generated always as identity primary key,
  destination_id bigint not null references destinations(id) on delete cascade,
  reviewer_name  text not null,
  rating         int  not null check (rating between 1 and 5),
  comment        text,
  created_at     timestamptz default now()
);

Expect "Success. No rows returned" and three tables in Table Editor. on delete cascade removes a destination's reviews with it; on delete set null keeps destinations when a category is deleted.

Phase 5 · Security (RLS and grants)

The browser uses a public key, so Row Level Security decides what is allowed: everyone can read, everyone can add a valid review, nobody can update or delete.

-- database/03_security.sql
alter table categories   enable row level security;
alter table destinations enable row level security;
alter table reviews      enable row level security;

create policy "read categories"   on categories   for select to anon, authenticated using (true);
create policy "read destinations" on destinations for select to anon, authenticated using (true);
create policy "read reviews"      on reviews      for select to anon, authenticated using (true);
create policy "add reviews" on reviews for insert to anon, authenticated
  with check (rating between 1 and 5
    and length(trim(reviewer_name)) between 1 and 50
    and length(coalesce(comment,'')) <= 500);

grant usage  on schema public to anon, authenticated;
grant select on categories, destinations, reviews to anon, authenticated;
grant insert on reviews to anon, authenticated;

Check Authentication → Policies shows 4 policies. Never disable RLS; re-run the script if data is missing.

Phase 6 · Recommendation view and queries

-- database/04_view.sql
create view destination_scores with (security_invoker = on) as
select d.*, c.name as category,
       round(avg(r.rating),1) as avg_rating,
       count(r.id) as review_count
from destinations d
left join categories c on c.id = d.category_id
left join reviews r on r.destination_id = d.id
group by d.id, c.name;

grant select on destination_scores to anon, authenticated;
select * from destination_scores order by avg_rating desc nulls last;

LEFT JOIN keeps destinations with no reviews. This view is the recommendation engine: the website filters it and sorts by rating.

Part A: run and screenshot each

-- database/05_queries.sql
-- Q1 destinations with category (JOIN)
select d.name, d.state, c.name as category from destinations d join categories c on c.id = d.category_id;
-- Q2 destinations per category (GROUP BY)
select c.name, count(d.id) from categories c left join destinations d on d.category_id = c.id group by c.name;
-- Q3 low budget places for winter
select name from destinations where budget = 'Low' and best_season = 'Winter';
-- Q4 average rating at least 4.5 (HAVING)
select d.name, round(avg(r.rating),1) from destinations d join reviews r on r.destination_id = d.id group by d.name having avg(r.rating) >= 4.5;
-- Q5 destinations with no reviews (subquery)
select name from destinations where id not in (select destination_id from reviews);
-- Q6 longest ideal trip
select name, ideal_days from destinations where ideal_days = (select max(ideal_days) from destinations);
-- Q7 update a description
update destinations set description = 'Tea estates, waterfalls and cool weather.' where name = 'Munnar';
-- Q8 top 3 recommendations for Beach lovers
select name, avg_rating from destination_scores where category = 'Beach' order by avg_rating desc nulls last limit 3;

app.js

const db = supabase.createClient(SUPABASE_URL, SUPABASE_KEY);
const $ = (id) => document.getElementById(id);
let places = [], currentId = null;

function esc(t) {  // blocks HTML/JS injection
  return String(t ?? "").replace(/&/g,"&amp;").replace(/</g,"&lt;").replace(/>/g,"&gt;").replace(/"/g,"&quot;").replace(/'/g,"&#39;");
}

async function loadCategories() {            // SELECT id, name FROM categories
  const { data, error } = await db.from("categories").select("id, name").order("name");
  if (error) return ($("status").textContent = "Error: " + error.message);
  data.forEach((c) => $("category").add(new Option(c.name, c.id)));
}

async function loadPlaces() {                // recommendation query
  $("status").textContent = "Loading...";
  let q = db.from("destination_scores").select("*")
    .order("avg_rating", { ascending: false, nullsFirst: false })
    .order("review_count", { ascending: false });
  if ($("search").value.trim()) {
    const s = $("search").value.trim();
    q = q.or(`name.ilike.%${s}%,state.ilike.%${s}%`);
  }
  if ($("category").value) q = q.eq("category_id", $("category").value);
  if ($("budget").value)   q = q.eq("budget", $("budget").value);
  if ($("season").value)   q = q.eq("best_season", $("season").value);
  const { data, error } = await q;
  if (error) return ($("status").textContent = "Error: " + error.message);
  places = data;
  $("status").textContent = data.length ? data.length + " recommendation(s), best rated first" : "No match. Try other filters 😕";
  $("grid").innerHTML = data.map((p) => `
    <div class="card" data-id="${p.id}">
      <img src="${esc(p.image_url)}" alt="${esc(p.name)}">
      <div><h3>${esc(p.name)}</h3>
      <p>${esc(p.state)} · ${esc(p.category)} · ${esc(p.budget)} budget</p>
      <p class="stars">${p.avg_rating ? "⭐ " + p.avg_rating + " (" + p.review_count + ")" : "No ratings yet"}</p></div>
    </div>`).join("");
  document.querySelectorAll(".card").forEach((c) => c.onclick = () => openPlace(Number(c.dataset.id)));
}

async function openPlace(id) {
  const p = places.find((x) => x.id === id); currentId = id;
  $("detail").innerHTML = `<h2>${esc(p.name)}</h2>
    <p>${esc(p.state)} · Best in ${esc(p.best_season)} · ${p.ideal_days} days</p><p>${esc(p.description)}</p>`;
  $("modal").showModal();
  loadReviews(id);
}

async function loadReviews(id) {             // SELECT ... FROM reviews WHERE destination_id = ?
  const { data, error } = await db.from("reviews").select("reviewer_name, rating, comment")
    .eq("destination_id", id).order("created_at", { ascending: false });
  $("reviews").innerHTML = error ? esc(error.message) : data.length
    ? data.map((r) => `<div class="review">${"⭐".repeat(r.rating)} ${esc(r.reviewer_name)}<p>${esc(r.comment)}</p></div>`).join("")
    : "No reviews yet.";
}

$("form").addEventListener("submit", async (e) => {   // INSERT INTO reviews
  e.preventDefault();
  const { error } = await db.from("reviews").insert({
    destination_id: currentId, reviewer_name: $("rname").value.trim(),
    rating: Number($("rrating").value), comment: $("rcomment").value.trim() });
  if (error) return alert("Could not save: " + error.message);
  $("form").reset(); await loadReviews(currentId); await loadPlaces();
});
